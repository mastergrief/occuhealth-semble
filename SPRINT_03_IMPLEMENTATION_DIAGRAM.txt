STATE MANAGEMENT PATTERNS
==========================

Pattern 1: Single Async Operation (Appointments, Settings Save)
──────────────────────────────────────────────────────────────
const [isLoading, setIsLoading] = useState(false);
const [error, setError] = useState<string | null>(null);

const handler = async () => {
  setIsLoading(true);
  setError(null);
  try {
    await mutation();
  } catch (err) {
    setError(extractMessage(err));
  } finally {
    setIsLoading(false);
  }
};


Pattern 2: Multiple Async Operations (Schedule: Add + Block)
────────────────────────────────────────────────────────────
const [op1Loading, setOp1Loading] = useState(false);
const [op2Loading, setOp2Loading] = useState<Id | null>(null);
const [op1Error, setOp1Error] = useState<string | null>(null);

// Each operation has independent state
const handleOp1 = async () => {
  setOp1Loading(true);
  try {
    await mutation1();
  } catch (err) {
    setOp1Error(extractMessage(err));
  } finally {
    setOp1Loading(false);
  }
};

const handleOp2 = async (id) => {
  setOp2Loading(id);
  try {
    await mutation2(id);
  } finally {
    setOp2Loading(null);
  }
};


Pattern 3: Multi-Step Mutation (Reports: Create + Send)
───────────────────────────────────────────────────────
const [isLoading, setIsLoading] = useState(false);
const [error, setError] = useState<string | null>(null);

const handler = async () => {
  setIsLoading(true);
  let intermediateResult = null;

  try {
    // Step 1
    intermediateResult = await mutation1();

    // Step 2 (uses Step 1 result)
    await mutation2(intermediateResult);

    // Success
  } catch (err) {
    // Can distinguish failure point using intermediateResult
    if (intermediateResult) {
      setError("Step 2 failed, Step 1 succeeded");
    } else {
      setError("Step 1 failed");
    }
  } finally {
    setIsLoading(false);
  }
};


Pattern 4: Status Tracking (Settings: Success/Error/Idle)
─────────────────────────────────────────────────────────
const [status, setStatus] = useState<"idle" | "success" | "error">("idle");
const [message, setMessage] = useState<string | null>(null);
const [isLoading, setIsLoading] = useState(false);

const handler = async () => {
  setIsLoading(true);
  setStatus("idle");
  setMessage(null);

  try {
    await mutation();
    setStatus("success");
    setTimeout(() => setStatus("idle"), 3000); // Auto-clear
  } catch (err) {
    setStatus("error");
    setMessage(extractMessage(err));
  } finally {
    setIsLoading(false);
  }
};
